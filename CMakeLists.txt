cmake_minimum_required(VERSION 3.18)
project(gpu_montecarlo VERSION 0.3.0 LANGUAGES CXX)
option(MC_ENABLE_OPENMP "Build the OpenMP backend" ON)
option(MC_BUILD_CPU_TOOLS "Build Windows CPU benchmark instrumentation" OFF)
option(MC_ENABLE_CUDA "Build full-device and explicit hybrid CUDA modes (at most 10000000 scenarios)" OFF)
configure_file(include/mc/version.hpp.in generated/mc/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mc_core src/core.cpp src/data.cpp src/output.cpp)
target_include_directories(mc_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(mc_core PRIVATE /W4)
else()
  target_compile_options(mc_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mc_var src/main.cpp)
target_link_libraries(mc_var PRIVATE mc_core)
if(MC_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  add_library(mc_openmp src/openmp.cpp)
  target_link_libraries(mc_openmp PUBLIC mc_core OpenMP::OpenMP_CXX)
  target_link_libraries(mc_var PRIVATE mc_openmp)
  target_compile_definitions(mc_var PRIVATE MC_HAS_OPENMP=1)
  if(MSVC)
    target_compile_options(mc_openmp PRIVATE /W4)
  else()
    target_compile_options(mc_openmp PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(MC_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  find_path(MC_CUB_INCLUDE_DIR cub/device/device_radix_sort.cuh
    HINTS ${CUDAToolkit_INCLUDE_DIRS} PATH_SUFFIXES cccl REQUIRED)
  add_library(mc_cuda src/cuda.cu)
  target_include_directories(mc_cuda PRIVATE "${MC_CUB_INCLUDE_DIR}")
  target_link_libraries(mc_cuda PUBLIC mc_core CUDA::cudart CUDA::curand)
  set_target_properties(mc_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Shared)
  target_compile_options(mc_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(mc_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/Zc:preprocessor>)
  endif()
  target_link_libraries(mc_var PRIVATE mc_cuda)
  target_compile_definitions(mc_var PRIVATE MC_HAS_CUDA=1)
  if(WIN32)
    find_file(MC_CUDART_DLL NAMES "cudart64_${CUDAToolkit_VERSION_MAJOR}.dll" cudart64_110.dll
      HINTS "${CUDAToolkit_BIN_DIR}" PATH_SUFFIXES x64 REQUIRED)
    find_file(MC_CURAND_DLL NAMES curand64_10.dll HINTS "${CUDAToolkit_BIN_DIR}" PATH_SUFFIXES x64 REQUIRED)
    function(mc_copy_cuda_runtime target)
      add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${MC_CUDART_DLL}" "${MC_CURAND_DLL}" "$<TARGET_FILE_DIR:${target}>")
    endfunction()
    mc_copy_cuda_runtime(mc_var)
  endif()
endif()
if(MC_BUILD_CPU_TOOLS)
  if(NOT WIN32 OR NOT MC_ENABLE_OPENMP)
    message(FATAL_ERROR "CPU instrumentation requires Windows and MC_ENABLE_OPENMP=ON")
  endif()
  add_executable(mc_cpu_probe tools/cpu_probe.cpp)
  target_link_libraries(mc_cpu_probe PRIVATE OpenMP::OpenMP_CXX PowrProf Version)
  if(MSVC)
    target_compile_options(mc_cpu_probe PRIVATE /W4)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mc_cuda_plan_tests tests/test_cuda_plan.cpp)
  target_link_libraries(mc_cuda_plan_tests PRIVATE mc_core)
  add_test(NAME cuda_plan_tests COMMAND mc_cuda_plan_tests)
  add_executable(mc_tests tests/test_core.cpp)
  target_link_libraries(mc_tests PRIVATE mc_core)
  add_test(NAME core_tests COMMAND mc_tests)
  add_executable(mc_output_tests tests/test_output.cpp)
  target_link_libraries(mc_output_tests PRIVATE mc_core)
  add_test(NAME json_output_tests COMMAND mc_output_tests)
  add_executable(mc_data_tests tests/test_data.cpp)
  target_link_libraries(mc_data_tests PRIVATE mc_core)
  add_test(NAME csv_calibration_tests COMMAND mc_data_tests)
  add_test(NAME csv_cli_smoke COMMAND mc_var --csv data/synthetic_prices.csv 10000 42 10)
  set_tests_properties(csv_cli_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME cli_contract_tests COMMAND "${CMAKE_COMMAND}"
    "-DCLI=$<TARGET_FILE:mc_var>" "-DOPENMP_ENABLED=${MC_ENABLE_OPENMP}" "-DCUDA_ENABLED=${MC_ENABLE_CUDA}"
    "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.cmake")
  if(MC_ENABLE_OPENMP)
    add_executable(mc_openmp_tests tests/test_openmp.cpp)
    target_link_libraries(mc_openmp_tests PRIVATE mc_openmp)
    add_test(NAME openmp_tests COMMAND mc_openmp_tests)
  endif()
  if(MC_ENABLE_CUDA)
    add_executable(mc_cuda_tests tests/test_cuda.cpp)
    target_link_libraries(mc_cuda_tests PRIVATE mc_cuda)
    if(WIN32)
      mc_copy_cuda_runtime(mc_cuda_tests)
    endif()
    add_test(NAME cuda_common_input_tests COMMAND mc_cuda_tests)
    set_tests_properties(cuda_common_input_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(mc_cuda_hybrid_tests tests/test_cuda_hybrid.cpp)
    target_link_libraries(mc_cuda_hybrid_tests PRIVATE mc_cuda)
    if(WIN32)
      mc_copy_cuda_runtime(mc_cuda_hybrid_tests)
    endif()
    add_test(NAME cuda_hybrid_tests COMMAND mc_cuda_hybrid_tests)
  endif()
  if(MC_BUILD_CPU_TOOLS)
    add_test(NAME cpu_recorder_statistics COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cpu_recorder.ps1")
    add_test(NAME cpu_probe_contract COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cpu_probe.ps1"
      -Probe "$<TARGET_FILE:mc_cpu_probe>" -Executable "$<TARGET_FILE:mc_var>")
  endif()
  if(WIN32)
    add_test(NAME demo_recorder_tests COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_demo_recorder.ps1"
      -Executable "$<TARGET_FILE:mc_var>" -OpenMPEnabled "${MC_ENABLE_OPENMP}")
    add_test(NAME equity_calendar_tests COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_equity_calendar.ps1")
    add_test(NAME equity_import_tests COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_equity_import.ps1"
      -Executable "$<TARGET_FILE:mc_var>" -OpenMPEnabled "${MC_ENABLE_OPENMP}" -CudaEnabled "${MC_ENABLE_CUDA}")
  endif()
endif()
